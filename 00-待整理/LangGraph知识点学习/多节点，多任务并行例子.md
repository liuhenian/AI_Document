``` python
from typing import TypedDict, Annotated, List, Dict, Any, Optional, Union
from langgraph.graph.message import add_messages
from langchain_core.messages import BaseMessage
from dataclasses import dataclass
import operator

# ---- 数据获取子任务定义 ----
@dataclass
class DataTask:
    """规划Agent拆解出的单个数据获取子任务"""
    task_id: str                     # 唯一ID，用于匹配结果
    source: str                      # 数据源：mysql / redis / victoriametrics / pyroscope / loki
    query_type: str                  # 操作类型：sql / promql / logql / pyroscope_query / redis_get
    query_params: Dict[str, Any]     # 查询参数，如SQL语句、PromQL表达式、时间范围等
    description: str                 # 任务自然语言描述，方便调试与日志
    timeout: int = 10                # 超时秒数
    retry: int = 1                   # 最大重试次数


# ---- 可视化图表配置 ----
@dataclass
class ChartConfig:
    """前端ECharts等图表库的配置描述"""
    chart_type: str                  # 图表类型：line, bar, pie, flamegraph 等
    title: str
    data: Union[Dict, List]          # 结构化数据，或指向时序数据的查询结果引用
    options: Dict[str, Any]          # 样式、坐标轴等ECharts option覆盖项
    description: str                 # 图表说明


# ---- 主状态 ----
class AgentState(TypedDict):
    # ---- 对话上下文 ----
    messages: Annotated[List[BaseMessage], add_messages]   # 完整对话历史，使用LangGraph内置追加规约器

    # ---- 当前用户提问（每次新问题会重置整个流程） ----
    user_query: str

    # ---- 规划阶段产物 ----
    plan_tasks: List[DataTask]                             # 由规划Agent生成的子任务列表，需并行执行

    # ---- 并行数据获取的结果聚合 ----
    # 使用自定义合并规约器：不同分支产生的结果会合并到一个字典中
    task_results: Annotated[Dict[str, Any], operator.ior]  # key=task_id, value=工具返回的数据或错误

    # ---- 聚合后的分析素材 ----
    aggregated_context: Optional[str]                      # 汇总所有 task_results 形成的文本摘要，供分析Agent使用
    # 保留原始结构化数据，便于可视化Agent直接读取
    raw_task_data: Optional[Dict[str, Any]]                # 与 task_results 相同，但在聚合节点整理后保留

    # ---- 分析结果 ----
    analysis_report: Optional[str]                         # 根因分析报告，Markdown格式

    # ---- 可视化图表 ----
    visualizations: List[ChartConfig]                      # 分析阶段或可视化Agent生成的图表配置列表

    # ---- 最终输出 ----
    final_response: Optional[str]                          # 最终返回给前端的完整答案，可包含图表占位符

    # ---- 控制与监控字段 ----
    next_stage: str                                        # 流程阶段标识：planning | executing | aggregating | analyzing | visualizing | finished | error
    errors: Annotated[List[str], operator.add]             # 全局错误收集，使用列表追加规约器
    retry_counts: Dict[str, int]                           # 各 task_id 的重试计数
```

以下通过一个完整示例，展示如何基于前文设计的 `AgentState` 构建 LangGraph 多节点工作流，并利用 `Send` API 实现动态并行数据获取。示例场景：用户提问 **“我的端口为什么丢包？”**。

---

## 0.1 图结构总览

```python
from langgraph.graph import StateGraph, END
from langgraph.types import Send, Command
from typing import List
import json
# 引入之前定义的 AgentState, DataTask, ChartConfig 等

builder = StateGraph(AgentState)

# 注册所有节点
builder.add_node("parse_input", parse_input_node)
builder.add_node("planner", planner_node)
builder.add_node("tool_executor", tool_executor_node)
builder.add_node("aggregator", aggregator_node)
builder.add_node("analyzer", analyzer_node)
builder.add_node("visualizer", visualizer_node)
builder.add_node("finalizer", finalizer_node)

# 静态边
builder.set_entry_point("parse_input")
builder.add_edge("parse_input", "planner")
builder.add_edge("planner", "aggregator")          # planner 并行派发后汇聚
builder.add_edge("aggregator", "analyzer")
builder.add_edge("analyzer", "visualizer")
builder.add_edge("visualizer", "finalizer")
builder.add_edge("finalizer", END)

graph = builder.compile()
```

关键点：
- `planner` 节点**不直接连接** `tool_executor`。它在内部通过 `Command(send=...)` 动态创建并行分支，分支的目标节点就是 `tool_executor`。
- `planner -> aggregator` 是一条静态边。当 `planner` 返回的 `Command` 中所有 `Send` 任务执行完毕，全局状态更新完成后，图自动沿此边进入 `aggregator`，从而实现“先并行执行，再汇聚”。

## 0.2 执行流程与 State 变化详解

假设初始输入（用户问题）已通过前端作为 `HumanMessage` 传入 `messages`。

### 0.2.1 步骤 0：初始状态
```python
state = {
    "messages": [HumanMessage(content="我的端口为什么丢包？")],
    "user_query": "",
    "plan_tasks": [],
    "task_results": {},             # 空字典
    "aggregated_context": None,
    "raw_task_data": None,
    "analysis_report": None,
    "visualizations": [],
    "final_response": None,
    "next_stage": "",
    "errors": [],
    "retry_counts": {},
}
```

---

### 0.2.2 步骤 1：`parse_input` 节点
**职责**：从最新消息中提取用户问题，写入 `user_query`，便于后续节点直接使用。

```python
def parse_input_node(state: AgentState) -> dict:
    last_msg = state["messages"][-1]
    return {
        "user_query": last_msg.content,
        "next_stage": "planning"
    }
```

**状态变化**：
```python
state["user_query"] = "我的端口为什么丢包？"
state["next_stage"] = "planning"
```

---

### 0.2.3 步骤 2：`planner` 节点（核心并行派发）
**职责**：调用 LLM，将用户问题拆解为一组独立的 `DataTask`，然后通过 `Command` 的 `send` 参数并行派发至 `tool_executor`。

```python
def planner_node(state: AgentState) -> Command:
    # 1. 用 LLM 生成计划（实际会调用模型，此处简化）
    plan_tasks = [
        DataTask(
            task_id="port_meta",
            source="mysql",
            query_type="sql",
            query_params={"query": "SELECT device, port, admin_status FROM port_info WHERE ...      "},
            description="获取丢包端口的元数据与配置"
        ),
        DataTask(
            task_id="port_drop_ts",
            source="victoriametrics",
            query_type="promql",
            query_params={"query": "rate(port_drop_packets{device='switch01',port='Gi0/1'}[5m])",
                          "start": "-30m", "end": "now"},
            description="查询端口丢包时序数据"
        ),
        DataTask(
            task_id="relevant_logs",
            source="loki",
            query_type="logql",
            query_params={"query": '{job="switch"} |= "port Gi0/1" |= "drop"',
                          "start": "-30m", "end": "now"},
            description="搜索包含丢包关键字的日志"
        ),
    ]

    # 2. 为每个任务创建 Send 指令，并行调用 tool_executor 节点
    sends = [
        Send("tool_executor", {"current_task": task})
        for task in plan_tasks
    ]

    # 3. 返回 Command：更新状态（plan_tasks + 阶段），并触发并行任务
    return Command(
        update={
            "plan_tasks": plan_tasks,
            "next_stage": "executing"
        },
        send=sends
    )
```

**状态变化**：
```python
state["plan_tasks"] = [DataTask(...), DataTask(...), DataTask(...)]   # 三个任务
state["next_stage"] = "executing"
# task_results 仍为 {}
```

**框架行为**：
- LangGraph 先将 `update` 写入全局状态。
- 然后立即并行启动三个 `tool_executor` 实例，每个实例接收一个仅包含 `current_task` 的状态片段。
- 这三个实例互不依赖，可同时查询 MySQL、VictoriaMetrics、Loki。

---

### 0.2.4 步骤 3：`tool_executor` 节点（并行执行）
**职责**：根据 `current_task` 的描述执行实际查询，并返回以 `task_id` 为键的结果（状态片段）。

```python
def tool_executor_node(state: AgentState) -> dict:
    task = state["current_task"]
    try:
        # 实际会路由到不同的数据连接器
        if task.source == "mysql":
            data = mysql_query(task.query_params)
        elif task.source == "victoriametrics":
            data = vm_query(task.query_params)
        elif task.source == "loki":
            data = loki_query(task.query_params)
        else:
            raise ValueError(f"Unknown source: {task.source}")
        result = {"status": "success", "data": data}
    except Exception as e:
        result = {"status": "error", "error_message": str(e)}
        # 可同时记录到 errors
        return {
            "task_results": {task.task_id: result},
            "errors": [f"{task.task_id} failed: {e}"]
        }

    return {
        "task_results": {task.task_id: result}
    }
```

**并行合并过程**：  
三个执行器实例可能先后返回如下状态片段：

```python
# 实例1（MySQL）
{"task_results": {"port_meta": {"status": "success", "data": {"device": "switch01", ...}}}}

# 实例2（VictoriaMetrics）
{"task_results": {"port_drop_ts": {"status": "success", "data": {"timestamps": [...], "values": [...]}}}}

# 实例3（Loki）
{"task_results": {"relevant_logs": {"status": "success", "data": ["2025-01-01 ... drop", ...]}}}
```

LangGraph 自动使用 `operator.ior` 规约器（在 `AgentState` 中定义）合并这些 `task_results` 字典，最终全局状态变为：

```python
state["task_results"] = {
    "port_meta":       {"status": "success", "data": {...}},
    "port_drop_ts":    {"status": "success", "data": {...}},
    "relevant_logs":   {"status": "success", "data": [...]}
}
```

**异常处理示例**：若 Loki 查询超时，对应返回片段可能是：
```python
{"task_results": {"relevant_logs": {"status": "error", "error_message": "timeout"}}}
```
该错误仍会被合并，后续分析 Agent 会感知到数据缺失。

---

### 0.2.5 步骤 4：`aggregator` 汇聚节点
**职责**：等所有 Send 完成后（框架自动等待），将 `task_results` 整理成 LLM 可读的文本摘要，并保留结构化数据供图表生成。

```python
def aggregator_node(state: AgentState) -> dict:
    raw = state["task_results"]
    # 生成文本摘要
    ctx_parts = []
    for tid, result in raw.items():
        if result["status"] == "success":
            # 根据数据源不同，生成不同格式的描述（简化示例）
            if tid == "port_meta":
                meta = result["data"]
                ctx_parts.append(f"端口元数据：设备 {meta['device']}, 端口 {meta['port']}, 管理状态 {meta['admin_status']}")
            elif tid == "port_drop_ts":
                ts_data = result["data"]
                avg_drop = sum(ts_data["values"]) / len(ts_data["values"])
                ctx_parts.append(f"最近30分钟平均丢包率：{avg_drop:.2f} pps，存在周期性尖峰。")
            elif tid == "relevant_logs":
                lines = result["data"][:5]   # 截取前5条
                ctx_parts.append("相关日志摘录：" + "; ".join(lines))
        else:
            ctx_parts.append(f"任务 {tid} 查询失败：{result.get('error_message', '未知错误')}")

    return {
        "aggregated_context": "\n".join(ctx_parts),
        "raw_task_data": raw,                     # 保留原始数据
        "next_stage": "analyzing"
    }
```

**状态变化**：
```python
state["aggregated_context"] = "端口元数据：设备 switch01... 平均丢包率 15.2 pps... 相关日志摘录：..."
state["raw_task_data"] = { ... }    # 与 task_results 相同
state["next_stage"] = "analyzing"
```

### 0.2.6 步骤 5：`analyzer` 分析节点
**职责**：基于聚合后的文本，调用 LLM 生成根因分析报告（Markdown 格式），并可指定需要生成的图表。

```python
def analyzer_node(state: AgentState) -> dict:
    prompt = f"""
    用户问题：{state['user_query']}
    收集到的数据摘要：
    {state['aggregated_context']}
    
    请分析根因，输出分析报告，并指出是否需要可视化图表（如果需要，请用标记 <chart> 描述）。
    """
    # 模拟 LLM 输出
    report = """
## 根因分析
端口 Gi0/1 丢包可能由以下原因导致：
1. **瞬时流量突发**：从时序数据可见每隔10分钟出现一次丢包尖峰，与日志中记录的 burst 关键字吻合。
2. **ACL 规则误配置**：元数据显示端口管理状态为 up，但绑定大量过滤规则，可能造成软件层丢包。
建议：检查 ACL 配置，并持续监控流量基线。
<chart type="line" title="端口丢包趋势" data_source="port_drop_ts" />
    """
    return {
        "analysis_report": report,
        "next_stage": "visualizing"
    }
```

**状态变化**：
```python
state["analysis_report"] = "## 根因分析\n端口 Gi0/1 ..."
state["next_stage"] = "visualizing"
```

### 0.2.7 步骤 6：`visualizer` 可视化节点
**职责**：解析报告中的 `<chart>` 标记，从 `raw_task_data` 提取对应数据，构造 `ChartConfig` 列表。

```python
def visualizer_node(state: AgentState) -> dict:
    charts = []
    raw = state["raw_task_data"]
    if "port_drop_ts" in raw and raw["port_drop_ts"]["status"] == "success":
        ts = raw["port_drop_ts"]["data"]
        charts.append(ChartConfig(
            chart_type="line",
            title="端口丢包趋势",
            data=ts,
            options={"xAxis": {"data": ts["timestamps"]}, "yAxis": {"name": "pps"}},
            description="最近30分钟丢包率变化"
        ))
    # 可以继续处理其他图表标记
    return {
        "visualizations": charts,
        "next_stage": "finalizing"
    }
```

**状态变化**：
```python
state["visualizations"] = [ChartConfig(...)]
```

---

### 0.2.8 步骤 7：`finalizer` 最终输出节点
**职责**：整合报告与图表配置，生成前端可渲染的最终答案，并追加到对话历史。

```python
def finalizer_node(state: AgentState) -> dict:
    final = state["analysis_report"]
    # 将 <chart> 标记转换为前端可识别的占位符，例如 :::chart{id="port_drop_ts_chart"}
    final = final.replace('<chart type="line" title="端口丢包趋势" data_source="port_drop_ts" />',
                          ':::chart{id="port_drop_ts_chart" type="line" title="端口丢包趋势"}')
    return {
        "final_response": final,
        "messages": [AIMessage(content=final)],
        "next_stage": "finished"
    }
```

**最终状态**：
```python
state["final_response"] = "## 根因分析\n端口 Gi0/1... :::chart{...}"
state["messages"] = [...原有消息, AIMessage(...)]
```

流程结束，前端可解析 `AIMessage` 中的 `:::chart` 占位符，结合 `visualizations` 渲染动态图表。

## 0.3 总结：并行与状态合并的魔法

本例通过 **一条静态边 + Command(send)** 完美实现动态并行：
- `planner` 决定任务数量（3个），无需修改图结构。
- 每个 `Send` 产生的状态片段（`task_results` 子集）被 **`operator.ior` 规约器**自动合并，完全杜绝了手动同步问题。
- 所有并行分支完成后，`aggregator` 才开始运行，天然保证了“先数据后分析”的顺序。

这种设计让团队可以轻松扩展数据源（新增一个 `Send` 即可），并保持分析逻辑的纯粹，非常符合 APM 场景下多源异构数据的实时关联诊断需求。
